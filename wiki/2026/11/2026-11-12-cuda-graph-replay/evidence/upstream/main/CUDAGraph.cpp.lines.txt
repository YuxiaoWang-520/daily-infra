278: void CUDAGraph::capture_end_pre() {
279:   auto stream = at::cuda::getCurrentCUDAStream();
280: 
281:   TORCH_CHECK(stream.stream() == capture_stream_.stream(),
282:               "Capture must end on the same stream it began on.");
283: 
284:   // Capture is over once cudaStreamEndCapture returns (success or failure).
285:   // Clear bookkeeping before propagating the return status so watchdog-side
286:   // checks cannot observe stale "capture active" state on error paths.
287:   cudaError_t endCaptureErr = cudaStreamEndCapture(capture_stream_, &graph_);
288:   if (endCaptureErr == cudaSuccess && graph_ != nullptr) {
289:     // Own the returned graph immediately so reset() destroys it if validation
290:     // below rejects the capture.
291:     has_graph_ = true;
292:   }
293:   const size_t invalid_capture_free_count =
294:       c10::cuda::CUDAGraphMemory::markCaptureEnd(capture_dev_, capture_id_);
295:   {
296:     std::unique_lock<std::mutex> lock(_currently_capturing_graphs_mutex);
297:     TORCH_CHECK(
298:         _currently_capturing_graphs.count(capture_id_),
299:         "capture_end() called before capture_begin().");
300:     _currently_capturing_graphs.erase(capture_id_);
301:   }
302: 
303:   // End pool allocation before checking the capture error. This ensures
304:   // captures_underway is cleaned up even if cudaStreamEndCapture failed
305:   // (e.g. due to an illegal operation during capture). These calls are
306:   // safe regardless of whether the capture succeeded — they simply
307:   // remove the pool routing entry added by beginAllocateToPool.
308:   c10::cuda::CUDACachingAllocator::endAllocateToPool(capture_dev_, mempool_id_);
309:   at::getHostAllocator(at::kCUDA)->end_allocate_to_pool(mempool_id_);
310:   // Allocation recording has stopped (even if endCaptureErr is a failure), so
311:   // reset() must not end the pool again.
312:   capturing_to_pool_ = false;
313: #if defined(USE_ROCM)
314:   at::cuda::releaseCaptureCublasWorkspaces(capture_id_);
315: #endif
316:   while (!captured_generator_states_.empty() &&
317:          !captured_generator_states_.back().generator_state) {
318:     captured_generator_states_.pop_back();
319:   }
320:   AT_CUDA_CHECK(endCaptureErr);
321:   TORCH_CHECK(
322:       invalid_capture_free_count == 0,
323:       "A CUDA graph allocation was freed from an invalid conditional capture "
324:       "scope. A conditional body may free its own allocations, and an "
325:       "ancestor may free a child's allocation, but a child may not free an "
326:       "ancestor or sibling allocation.");
327: 
328:   TORCH_CHECK(graph_ != nullptr, "Invalid capture.");
329: 
330:   for (auto& captured_state : captured_generator_states_) {
331:     if (captured_state.capture_id == capture_id_) {
332:       captured_state.wholegraph_increment = captured_state.capture_state->finalize();
333:     }
334:   }
335: 
336:   size_t numCUDAGraphNodes = 0;
337:   AT_CUDA_CHECK(cudaGraphGetNodes(graph_, nullptr, &numCUDAGraphNodes));
338:   if (numCUDAGraphNodes == 0) {
339:       TORCH_WARN("The CUDA Graph is empty. This usually means that the graph was ",
340:                  "attempted to be captured on wrong device or stream.");
341:   }
342: 
343:   capture_ended_ = true;
344: }

364: void CUDAGraph::instantiate() {
365:   TORCH_CHECK(capture_ended_, "capture_end() must have been called before calling instantiate");
366: 
367:   if (has_graph_exec_) {
368:     TORCH_CHECK(keep_graph_, "instantiate() is intended to be called by the user only when keep_graph=true");
369:     AT_CUDA_CHECK(cudaGraphExecDestroy(graph_exec_));
370:   }
371:   // In typical graph usage some tensors (e.g. the tensors used for graph IO) are not freed
372:   // between replays.
373:   // If Pytorch compiles and runs with a CUDA 11.4+ toolkit, there's a chance the allocator backend
374:   // is cudaMallocAsync.
375:   // cudaMallocAsync is generally graph-safe, but if some tensors are not freed between replays,
376:   // the graph's internal bookkeeping requires that we instantiate with
377:   // cudaGraphInstantiateFlagAutoFreeOnLaunch. See
378:   // cudaGraphLaunch
379:   // https://docs.nvidia.com/cuda/cuda-runtime-api/group__CUDART__GRAPH.html#group__CUDART__GRAPH_1g1accfe1da0c605a577c22d9751a09597
380:   // cudaGraphInstantiateWithFlags
381:   // https://docs.nvidia.com/cuda/cuda-runtime-api/group__CUDART__GRAPH.html#group__CUDART__GRAPH_1ga2c652a24ba93e52b99a47bec0888233
382: #if !defined(USE_ROCM)
383:     AT_CUDA_CHECK(cudaGraphInstantiateWithFlags(&graph_exec_,
384:                                                 graph_,
385:                                                 cudaGraphInstantiateFlagAutoFreeOnLaunch | cudaGraphInstantiateFlagUseNodePriority));
386: #else
387:     AT_CUDA_CHECK(cudaGraphInstantiateWithFlags(&graph_exec_,
388:                                                 graph_,
389:                                                 cudaGraphInstantiateFlagAutoFreeOnLaunch));
390: #endif
391:   has_graph_exec_ = true;
392: }
