669:     def capture_end(self) -> None:
670:         r"""End CUDA graph capture on the current stream.
671: 
672:         After ``capture_end``, ``replay`` may be called on this instance.
673: 
674:         Typically, you shouldn't call ``capture_end`` yourself.
675:         Use :class:`~torch.cuda.graph` or :func:`~torch.cuda.make_graphed_callables`,
676:         which call ``capture_end`` internally.
677:         """
678:         self.capture_end_pre()
679:         # Run the capture-end hooks while the template is live (both keep_graph modes). The
680:         # capture graph id is NOT read here: maybe_stamp_capture_root already stamped it at
681:         # capture_begin, and the template keeps that id for its whole life. Errors are user
682:         # bugs and propagate -- we deliberately don't wrap them in a finally that calls
683:         # capture_end_post(), since a failing finalize would mask the real error.
684:         if _global_capture_end_hooks:
685:             _run_global_hooks(_global_capture_end_hooks, self)
686:         for hook in list(self._capture_end_hooks.values()):
687:             hook(self)
688:         if not self._keep_graph:
689:             # Route keep_graph=False instantiation through Python so the remap
690:             # and post-instantiate hooks fire from instantiate().
691:             self.instantiate()
